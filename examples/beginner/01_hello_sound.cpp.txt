// The building blocks: the board (Bela.h), the sensors, and the
// synthesis, effects, filters and utilities. You do not need to
// change these.
#include <Bela.h>
#include "sensors/sensors.h"
#include "dsp/dsp.h"

// ---------------------------------------------------------------------
// 1. Sensors
// ---------------------------------------------------------------------

// No sensors yet - you'll add one in step 2.

// ---------------------------------------------------------------------
// 2. Sound
// ---------------------------------------------------------------------

// An oscillator makes a tone: a wave that repeats over and over.
// "dsp::Oscillator" is the kind of thing, "osc" is the name we give
// it - you could call it anything.
dsp::Oscillator osc;

// ---------------------------------------------------------------------
// 3. render()
// ---------------------------------------------------------------------

// The board calls this over and over, very fast, to ask "what's the
// next bit of sound?". Sound is a long list of numbers between -1 and
// 1 (called samples) - about 44100 of them every second.
void render(BelaContext *context, void *userData)
{
	// The board asks for a few samples at a time -
	// context->audioFrames of them - so work them out one by one.
	// "n" counts them: 0, 1, 2, ...
	for(unsigned int n = 0; n < context->audioFrames; n++)
	{
		// Ask the oscillator for its next sample
		float sample = osc.Process();

		// Send it to the left (0) and right (1) outputs
		audioWrite(context, n, 0, sample);
		audioWrite(context, n, 1, sample);
	}
}

// ---------------------------------------------------------------------
// 4. setup()
// ---------------------------------------------------------------------

// setup() runs once, when the program starts - before any sound.
// After it, the board keeps calling render() until you stop it.
bool setup(BelaContext *context, void *userData)
{
	// Keeps fading sounds from overloading the CPU - leave this here
	dsp::FlushDenormals();

	// Set up the oscillator. It needs to know the sample rate (how
	// many samples per second) to play at the right pitch.
	osc.Init(context->audioSampleRate);

	// The shape of the wave - this changes the sound's color
	osc.SetWaveform(dsp::Oscillator::WAVE_SINE);

	// The pitch, in Hz (vibrations per second). 220 Hz is the note A3.
	osc.SetFreq(220.f);

	// Start the note. It keeps playing until NoteOff() - which we
	// never call, so it plays until you stop the program.
	osc.NoteOn();

	// true = everything is ready, start the sound
	return true;
}

// Runs once when the program stops
void cleanup(BelaContext *context, void *userData)
{
	// Nothing to do here
}
