/** SMC26 DSP course example: random robot (sample and hold).
 *
 *  Sample and hold grabs a value at one moment and keeps it until the
 *  next grab - turning anything into a staircase of steps. Here it
 *  picks a new note on every step, and each step plays a short,
 *  plucky note: the bleepy "computer thinking" sound of old sci-fi
 *  films and analog synths.
 *
 *  Hold playButton to play. Press sourceButton to switch what the
 *  sample and hold grabs:
 *    1. random - a new random note every step (the classic sound)
 *    2. LFO    - grabs a slow wave as it moves, so the notes climb
 *                up and walk back down like a staircase
 *
 *  rateKnob     how many steps (notes) per second
 *  rangeKnob    how far the notes can jump: one note to two octaves
 *  cutoffKnob   the filter: low = dark and soft, high = bright
 *  lfoRateKnob  how fast the staircase climbs (LFO mode only)
 *  lengthKnob   how long each note rings: short blip to longer pluck
 *
 *  To try it, copy this whole file into render.cpp and type
 *  "make run" in the terminal. Ctrl-C stops it.
 *
 *  Wiring:
 *    playButton   - D0     rateKnob    - A0
 *    sourceButton - D1     rangeKnob   - A1
 *                          cutoffKnob  - A2
 *                          lfoRateKnob - A3
 *                          lengthKnob  - A4
 *  Each button connects its pin to 3.3V when pressed, with a 10k
 *  resistor from the pin to GND (a pull-down).
 */

// The building blocks: the board (Bela.h), the sensors, and the
// synthesis, effects, filters and utilities. You do not need to
// change these.
#include <Bela.h>
#include "sensors/sensors.h"
#include "dsp/dsp.h"

// ---------------------------------------------------------------------
// 1. Sensors
// ---------------------------------------------------------------------

Button playButton;
Button sourceButton;

Potentiometer rateKnob;
Potentiometer rangeKnob;
Potentiometer cutoffKnob;
Potentiometer lfoRateKnob;
Potentiometer lengthKnob;

// Change false to true to print the sensor values in the terminal
bool gPrintValues = false;

// Prints every sensor's value on one line. Add a line here for each
// sensor you add. Only runs when gPrintValues is true.
void PrintValues(BelaContext *context)
{
	PrintValue("play", playButton.IsPressed());
	PrintValue("lfoMode", sourceButton.Toggled());
	PrintValue("rate", dsp::FullRange(rateKnob.Read(context)));
	PrintValue("range", dsp::FullRange(rangeKnob.Read(context)));
	PrintValue("cutoff", dsp::FullRange(cutoffKnob.Read(context)));
	PrintValue("lfoRate", dsp::FullRange(lfoRateKnob.Read(context)));
	PrintValue("length", dsp::FullRange(lengthKnob.Read(context)));
	PrintLineEnd();
}

// ---------------------------------------------------------------------
// 2. Sound
// ---------------------------------------------------------------------

// The synth that plays the notes: waveform -> filter -> envelope
dsp::Subtractive synth;

// The sample and hold. Its built-in clock decides when to grab a new
// value (SetRate), and Stepped() tells us when that just happened.
dsp::SampleHold steps;

// The slow wave the sample and hold grabs in LFO mode
dsp::Lfo lfo;

// The note everything is centered on
float gRootNote = 220.f; // A3

// ---------------------------------------------------------------------
// 3. render()
// ---------------------------------------------------------------------

// Called by the board over and over, very fast, each time for the
// next few samples of sound. Keep it quick: no printf() and no waiting
// in here, or the sound will click and drop out.
void render(BelaContext *context, void *userData)
{
	// Read the buttons first, so JustPressed() and JustReleased()
	// below see the latest change
	playButton.Update(context);
	sourceButton.Update(context);

	// Reset() restarts the step clock, so the first note plays the
	// moment you press - not whenever the next step was due
	if(playButton.JustPressed())
		steps.Reset();
	if(playButton.JustReleased())
		synth.NoteOff();

	// Steps per second: from 1 (slow, one note at a time) to 16 (a
	// fast, chattering stream of notes)
	steps.SetRate(dsp::MapExponential(dsp::FullRange(rateKnob.Read(context)), 1.f, 16.f));

	// How far notes can jump: from 1 semitone to 2 octaves
	float noteRange = dsp::MapLinear(dsp::FullRange(rangeKnob.Read(context)), 1.f, 24.f);

	// Filter: MapExponential so the knob feels even across its whole
	// travel (see dsp/Utility.h)
	synth.SetCutoff(dsp::MapExponential(dsp::FullRange(cutoffKnob.Read(context)), 150.f, 4000.f));

	// The staircase speed: from one climb every 20 seconds to two a
	// second
	lfo.SetFreq(dsp::MapExponential(dsp::FullRange(lfoRateKnob.Read(context)), 0.05f, 2.f));

	// Note length: from a 30 ms blip to a 0.8 s pluck
	synth.SetAmpDecay(dsp::MapExponential(dsp::FullRange(lengthKnob.Read(context)), 0.03f, 0.8f));

	// About 10 times a second, print the sensor values - only if
	// gPrintValues is true
	if(gPrintValues && TimeToPrint(context))
		PrintValues(context);

	// Runs once per sample - context->audioFrames samples each time
	for(unsigned int n = 0; n < context->audioFrames; n++)
	{
		// The LFO keeps moving all the time, even when nobody listens
		float lfoValue = lfo.Process();

		// Ask the sample and hold for its value, -1 to 1. It only
		// changes on a step - in between, it repeats the held value.
		//   Process()         grabs a random value
		//   Process(lfoValue) grabs wherever the LFO is right now
		float held;
		if(sourceButton.Toggled())
			held = steps.Process(lfoValue);
		else
			held = steps.Process();

		// On every step, while playButton is held: a new note.
		// MapSemitones wants 0 - 1, so (held + 1) / 2 moves -1..1 into
		// that range, then snaps it to the A minor scale.
		if(steps.Stepped() && playButton.IsPressed())
		{
			float position = (held + 1.f) * 0.5f;
			synth.SetFreq(dsp::MapSemitones(position, gRootNote, noteRange, dsp::SCALE_MINOR));
			synth.NoteOn();
		}

		// The next sample of the synth, turned down by 6 dB to leave
		// some headroom
		float sample = dsp::ApplyGain(synth.Process(), -6.f);

		// Send it to the left (0) and right (1) outputs
		audioWrite(context, n, 0, sample);
		audioWrite(context, n, 1, sample);
	}
}

// ---------------------------------------------------------------------
// 4. setup()
// ---------------------------------------------------------------------

// Runs once, before the sound starts
bool setup(BelaContext *context, void *userData)
{
	// Keeps fading sounds from overloading the CPU - leave this here
	dsp::FlushDenormals();

	// Tell each sensor which pin it is plugged into
	playButton.Init(context, 0);   // digital pin D0
	sourceButton.Init(context, 1); // digital pin D1
	rateKnob.Init(0);              // analog input A0
	rangeKnob.Init(1);             // analog input A1
	cutoffKnob.Init(2);            // analog input A2
	lfoRateKnob.Init(3);           // analog input A3
	lengthKnob.Init(4);            // analog input A4

	float sampleRate = context->audioSampleRate;

	// Each note is a pluck: instant attack, then it dies away over the
	// decay (set by lengthKnob). Sustain 0 means it never holds, so
	// every step sounds like a separate note. The filter envelope adds
	// a bright "blip" at the start of each one.
	synth.Init(sampleRate);
	synth.SetWaveform(dsp::Subtractive::WAVE_SQUARE);
	synth.SetAmpAttack(0.002f);
	synth.SetAmpSustain(0.f);
	synth.SetAmpRelease(0.05f);
	synth.SetFilterEnvAmount(2000.f);
	synth.SetFilterDecay(0.08f);
	synth.SetResonance(0.5f);

	steps.Init(sampleRate);

	lfo.Init(sampleRate);
	lfo.SetWaveform(dsp::Lfo::WAVE_TRIANGLE);

	// From here on, the board keeps calling render()
	return true;
}

// Runs once when the program stops
void cleanup(BelaContext *context, void *userData)
{
	// Nothing to do here
}
