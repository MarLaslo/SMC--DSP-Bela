/** SMC26 DSP course example: wind and ocean.
 *
 *  Nature sounds made from noise. Noise is every pitch at once - a
 *  "shhhh" - and a filter picks out the part of it you want to hear.
 *  Move that filter slowly and the noise comes alive:
 *    - wind:  a narrow bandpass filter lets through one "howling"
 *             band of noise, and its pitch and volume drift around
 *             in uneven gusts
 *    - ocean: a lowpass filter, rumbly when a wave is far away and
 *             brighter and louder as it rolls in, again and again
 *  Both are the same few parts - noise, a filter and slow LFOs - just
 *  set up differently.
 *
 *  Press modeButton to switch between wind and ocean (each press
 *  flips it). Hold gustButton for a strong gust / a big wave.
 *
 *  speedKnob     how fast the gusts / waves come
 *  stormKnob     calm (gentle, steady) up to stormy (big swells)
 *  toneKnob      the pitch of the noise, low and rumbly to high and hissy
 *  howlKnob      filter resonance - turn up for a whistling, howling wind
 *  volumeKnob    overall volume
 *
 *  Sounds best in stereo headphones - each ear gets its own noise.
 *
 *  To try it, copy this whole file into render.cpp and type
 *  "make run" in the terminal. Ctrl-C stops it.
 *
 *  Wiring:
 *    gustButton - D0     speedKnob  - A0
 *    modeButton - D1     stormKnob  - A1
 *                        toneKnob   - A2
 *                        howlKnob   - A3
 *                        volumeKnob - A4
 *  Each button connects its pin to 3.3V when pressed, with a 10k
 *  resistor from the pin to GND (a pull-down).
 */

// The building blocks: the board (Bela.h), the sensors, and the
// synthesis, effects, filters and utilities. You do not need to
// change these.
#include <Bela.h>
#include <cmath>
#include "sensors/sensors.h"
#include "dsp/dsp.h"

// ---------------------------------------------------------------------
// 1. Sensors
// ---------------------------------------------------------------------

Button gustButton;
Button modeButton;

Potentiometer speedKnob;
Potentiometer stormKnob;
Potentiometer toneKnob;
Potentiometer howlKnob;
Potentiometer volumeKnob;

// Change false to true to print the sensor values in the terminal
bool gPrintValues = false;

// Prints every sensor's value on one line. Add a line here for each
// sensor you add. Only runs when gPrintValues is true.
void PrintValues(BelaContext *context)
{
	PrintValue("gust", gustButton.IsPressed());
	PrintValue("ocean", modeButton.Toggled());
	PrintValue("speed", dsp::FullRange(speedKnob.Read(context)));
	PrintValue("storm", dsp::FullRange(stormKnob.Read(context)));
	PrintValue("tone", dsp::FullRange(toneKnob.Read(context)));
	PrintValue("howl", dsp::FullRange(howlKnob.Read(context)));
	PrintValue("volume", dsp::FullRange(volumeKnob.Read(context)));
	PrintLineEnd();
}

// ---------------------------------------------------------------------
// 2. Sound
// ---------------------------------------------------------------------

// The raw sound: white noise
dsp::Noise noise;

// One filter for each ear. They get the same settings, but different
// noise, which makes the sound wide instead of stuck in the middle.
dsp::StateVariableFilter filterL;
dsp::StateVariableFilter filterR;

// Two slow LFOs that move the filter and volume. In wind mode they
// are mixed together: because their speeds don't match, their ups
// and downs never line up the same way twice - so the gusts feel
// random. In ocean mode only the first one is used, for steady,
// regular waves.
dsp::Lfo lfo1;
dsp::Lfo lfo2;

// Turns the gust button's sudden on/off into a smooth swell. A
// lowpass filter on a control value makes it glide instead of jump.
dsp::OnePoleFilter gustSmoother;

// Safety net on the final sound, so a big gust never distorts
dsp::Limiter limiterL;
dsp::Limiter limiterR;

// ---------------------------------------------------------------------
// 3. render()
// ---------------------------------------------------------------------

// Called by the board over and over, very fast, each time for the
// next few samples of sound. Keep it quick: no printf() and no waiting
// in here, or the sound will click and drop out.
void render(BelaContext *context, void *userData)
{
	// Read the buttons first, so they see the latest press
	gustButton.Update(context);
	modeButton.Update(context);

	// Toggled() flips on every press: starts false, so wind first
	bool oceanMode = modeButton.Toggled();

	// The only difference between the two sounds: which part of the
	// noise the filter lets through. Bandpass = just a band around the
	// cutoff (howling wind). Lowpass = everything below it (rumbling
	// sea).
	if(oceanMode)
	{
		filterL.SetMode(dsp::StateVariableFilter::MODE_LOWPASS);
		filterR.SetMode(dsp::StateVariableFilter::MODE_LOWPASS);
	}
	else
	{
		filterL.SetMode(dsp::StateVariableFilter::MODE_BANDPASS);
		filterR.SetMode(dsp::StateVariableFilter::MODE_BANDPASS);
	}

	// 1 while the button is held, 0 when it's not - the smoother turns
	// that into a gentle swell, gliding between 0 and 1. It runs once
	// per render(), not once per sample: that's plenty for something
	// this slow (see setup()).
	float gustTarget;
	if(gustButton.IsPressed())
		gustTarget = 1.f;
	else
		gustTarget = 0.f;
	float gust = gustSmoother.Process(gustTarget);

	// Speed: from one wave every 20 seconds (0.05 Hz) up to two a
	// second. lfo2 runs at a bit over a third of lfo1's speed - an
	// uneven ratio, so the two never fall into step.
	float speed = dsp::MapExponential(dsp::FullRange(speedKnob.Read(context)), 0.05f, 2.f);
	lfo1.SetFreq(speed);
	lfo2.SetFreq(speed * 0.37f);

	// Storm is already 0 - 1, same as the knob
	float storm = dsp::FullRange(stormKnob.Read(context));

	// Tone: MapExponential so the knob feels even across its whole
	// travel (see dsp/Utility.h). Hz, the filter's center when calm.
	float baseTone = dsp::MapExponential(dsp::FullRange(toneKnob.Read(context)), 150.f, 3000.f);

	// Howl stops at 0.9 - above that the filter starts to ring on its
	// own
	float howl = dsp::MapLinear(dsp::FullRange(howlKnob.Read(context)), 0.f, 0.9f);
	filterL.SetResonance(howl);
	filterR.SetResonance(howl);

	// Volume from -30 dB (quiet) to +6 dB (loud)
	float volumeDb = dsp::MapLinear(dsp::FullRange(volumeKnob.Read(context)), -30.f, 6.f);

	// About 10 times a second, print the sensor values - only if
	// gPrintValues is true
	if(gPrintValues && TimeToPrint(context))
		PrintValues(context);

	// Runs once per sample - context->audioFrames samples each time
	for(unsigned int n = 0; n < context->audioFrames; n++)
	{
		// Where the slow movement is right now, from -1 to 1
		float movement;
		if(oceanMode)
			movement = lfo1.Process();
		else
			movement = 0.6f * lfo1.Process() + 0.4f * lfo2.Process();

		// Filter: moves up to 2 octaves either way with the movement
		// (more when stormy), plus up to 1.5 octaves higher in a gust.
		// Each octave doubles the frequency, hence powf(2, ...).
		float octaves = storm * 2.f * movement + gust * 1.5f;
		float cutoff = dsp::Clamp(baseTone * powf(2.f, octaves), 40.f, 12000.f);
		filterL.SetFreq(cutoff);
		filterR.SetFreq(cutoff);

		// Volume swells with the movement too. "swell" is the movement
		// moved from -1..1 to 0..1. When calm the volume barely moves;
		// when stormy it drops right down between gusts.
		float swell = (movement + 1.f) / 2.f;
		float level = dsp::MapLinear(swell, 1.f - storm, 1.f) + gust;

		// Ask the noise for two different random samples, one for each
		// ear, and filter each one
		float left = filterL.Process(noise.Process()) * level;
		float right = filterR.Process(noise.Process()) * level;

		// Overall volume, then the safety net
		left = limiterL.Process(dsp::ApplyGain(left, volumeDb));
		right = limiterR.Process(dsp::ApplyGain(right, volumeDb));

		// Each ear its own: left (0) and right (1)
		audioWrite(context, n, 0, left);
		audioWrite(context, n, 1, right);
	}
}

// ---------------------------------------------------------------------
// 4. setup()
// ---------------------------------------------------------------------

// Runs once, before the sound starts
bool setup(BelaContext *context, void *userData)
{
	// Keeps fading sounds from overloading the CPU - leave this here
	dsp::FlushDenormals();

	// Tell each sensor which pin it is plugged into
	gustButton.Init(context, 0); // digital pin D0
	modeButton.Init(context, 1); // digital pin D1
	speedKnob.Init(0);           // analog input A0
	stormKnob.Init(1);           // analog input A1
	toneKnob.Init(2);            // analog input A2
	howlKnob.Init(3);            // analog input A3
	volumeKnob.Init(4);          // analog input A4

	float sampleRate = context->audioSampleRate;

	noise.Init();
	filterL.Init(sampleRate);
	filterR.Init(sampleRate);
	lfo1.Init(sampleRate);
	lfo2.Init(sampleRate);
	limiterL.Init();
	limiterR.Init();

	// The smoother runs once per render(), so it gets the render()
	// rate instead of the sample rate: sampleRate / audioFrames times
	// a second. 0.5 Hz: the gust swells in, and dies away, over about
	// a second. Lower it for slower swells.
	gustSmoother.Init(sampleRate / context->audioFrames);
	gustSmoother.SetFreq(0.5f);

	// From here on, the board keeps calling render()
	return true;
}

// Runs once when the program stops
void cleanup(BelaContext *context, void *userData)
{
	// Nothing to do here
}
