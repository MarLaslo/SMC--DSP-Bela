/** SMC26 DSP course example: voice manager.
 *
 *  The simplest way to play more than one note at a time. A synth
 *  voice (like dsp::Subtractive) can only play one note. To play a
 *  chord you need several voices - dsp::VoiceManager holds them for
 *  you and hands each new note to a free one.
 *
 *  Hold button1 to play an A, hold button2 to play an E. Hold both
 *  and you hear both notes together. cutoffKnob makes every voice
 *  darker or brighter. bendKnob bends the pitch of every note up or
 *  down by up to 2 semitones (knob in the middle = no bend).
 *
 *  To try it, copy this whole file into render.cpp and type
 *  "make run" in the terminal. Ctrl-C stops it.
 *
 *  Wiring:
 *    button1 - D0     cutoffKnob - A0
 *    button2 - D1     bendKnob   - A1
 *  Each button connects its pin to 3.3V when pressed, with a 10k
 *  resistor from the pin to GND (a pull-down).
 */

// The building blocks: the board (Bela.h), the sensors, and the
// synthesis, effects, filters and utilities. You do not need to
// change these.
#include <Bela.h>
#include "sensors/sensors.h"
#include "dsp/dsp.h"

// ---------------------------------------------------------------------
// 1. Sensors
// ---------------------------------------------------------------------

Button button1;
Button button2;

Potentiometer cutoffKnob;
Potentiometer bendKnob;

// Change false to true to print the sensor values in the terminal
bool gPrintValues = false;

// Prints every sensor's value on one line. Add a line here for each
// sensor you add. Only runs when gPrintValues is true.
void PrintValues(BelaContext *context)
{
	PrintValue("button1", button1.IsPressed());
	PrintValue("button2", button2.IsPressed());
	PrintValue("cutoff", dsp::FullRange(cutoffKnob.Read(context)));
	PrintValue("bend", dsp::FullRange(bendKnob.Read(context)));
	PrintLineEnd();
}

// ---------------------------------------------------------------------
// 2. Sound
// ---------------------------------------------------------------------

// How many notes can play at the same time. With 2 voices and 2
// buttons, every button always gets its own voice.
const int kNumVoices = 2;

// The voice manager: kNumVoices copies of dsp::Subtractive in one
// object. Swap dsp::Subtractive for another synth from
// dsp/Synthesis (e.g. dsp::FMSimple, dsp::ModalVoice) to change the
// sound - the rest of the code stays the same.
dsp::VoiceManager<dsp::Subtractive, kNumVoices> voices;

// The note each button plays - note names instead of Hz, see
// dsp::NoteToFreq in dsp/Utility.h
float gNote1 = dsp::NoteToFreq("A3");
float gNote2 = dsp::NoteToFreq("E4");

// ---------------------------------------------------------------------
// 3. render()
// ---------------------------------------------------------------------

// Called by the board over and over, very fast, each time for the
// next few samples of sound. Keep it quick: no printf() and no waiting
// in here, or the sound will click and drop out.
void render(BelaContext *context, void *userData)
{
	// Read the buttons first, so JustPressed() and JustReleased()
	// below see the latest change
	button1.Update(context);
	button2.Update(context);

	// Button goes down -> start its note. Button comes up -> stop it.
	// NoteOff() needs the same frequency as NoteOn(), so the voice
	// manager knows which voice to stop.
	if(button1.JustPressed())
		voices.NoteOn(gNote1);
	if(button1.JustReleased())
		voices.NoteOff(gNote1);

	if(button2.JustPressed())
		voices.NoteOn(gNote2);
	if(button2.JustReleased())
		voices.NoteOff(gNote2);

	// Settings like the cutoff belong to each voice, not to the voice
	// manager. voices.Voice(v) gives you voice number v, so loop over
	// all of them and give each the same setting.
	float cutoff = dsp::MapExponential(dsp::FullRange(cutoffKnob.Read(context)), 200.f, 5000.f);
	for(int v = 0; v < kNumVoices; v++)
		voices.Voice(v).SetCutoff(cutoff);

	// Pitch bend. voices.Freq(v) is the note voice v was given by
	// NoteOn() - it never changes, so we always bend from the original
	// note. MapPitchBend turns the knob into a frequency up to 2
	// semitones below (knob down) or above (knob up) that note, and
	// Voice(v).SetFreq() makes the voice play it.
	float bend = dsp::FullRange(bendKnob.Read(context));
	for(int v = 0; v < kNumVoices; v++)
		voices.Voice(v).SetFreq(dsp::MapPitchBend(bend, voices.Freq(v), 2.f));

	// About 10 times a second, print the sensor values - only if
	// gPrintValues is true
	if(gPrintValues && TimeToPrint(context))
		PrintValues(context);

	// Runs once per sample - context->audioFrames samples each time
	for(unsigned int n = 0; n < context->audioFrames; n++)
	{
		// Process() adds all the voices together into one sample.
		// Two loud voices added together can go over 1 and distort,
		// so divide by the number of voices to keep it clean.
		float sample = voices.Process() / kNumVoices;

		// Send it to the left (0) and right (1) outputs
		audioWrite(context, n, 0, sample);
		audioWrite(context, n, 1, sample);
	}
}

// ---------------------------------------------------------------------
// 4. setup()
// ---------------------------------------------------------------------

// Runs once, before the sound starts
bool setup(BelaContext *context, void *userData)
{
	// Keeps fading sounds from overloading the CPU - leave this here
	dsp::FlushDenormals();

	// Tell each sensor which pin it is plugged into
	button1.Init(context, 0); // digital pin D0
	button2.Init(context, 1); // digital pin D1
	cutoffKnob.Init(0);       // analog input A0
	bendKnob.Init(1);         // analog input A1

	// Sets up every voice at once - they need to know the sample rate
	voices.Init(context->audioSampleRate);

	// From here on, the board keeps calling render()
	return true;
}

// Runs once when the program stops
void cleanup(BelaContext *context, void *userData)
{
	// Nothing to do here
}
